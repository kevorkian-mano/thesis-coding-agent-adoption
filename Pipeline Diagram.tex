\node[phase=c2bg, below=0.55cm of p1] (p2) {%

% ── Phase 3 ──────────────────────────────────────────────────────────────────
\node[phase=c3bg, below=0.55cm of p2] (p3) {%
  \textbf{\textcolor{c3bd}{\large Phase 3 \quad AI Adoption Detection (3 Passes)}}\\[5pt]
  $\bullet$ \ \textbf{Pass 1} -- config file check, batch GraphQL (10 repos/request):
             \texttt{CLAUDE.md}, \texttt{.cursorrules}, \texttt{GEMINI.md},
             \texttt{copilot-instructions.md}, \texttt{.aider.conf.yml} \ldots
             $\rightarrow$ match = \textbf{agentic}\\[3pt]
  $\bullet$ \ \textbf{Pass 2} -- 100 commits + 50 PRs:
             \texttt{Co-authored-by:} trailer patterns, PR branch prefixes
             (\texttt{copilot/}, \texttt{devin/}, \texttt{codex/})\\[3pt]
  $\bullet$ \ \textbf{Pass 3} -- 300 commits + 100 PRs + \texttt{package.json}
             SDK deps: full signal scan on remaining unclassified repos\\[4pt]
  \textbf{\textcolor{c3bd}{Rule:}} \
  $\geq$3 signals or $\geq$2 agents $\Rightarrow$ \textbf{agentic} \quad
  1--2 signals, 1 agent $\Rightarrow$ \textbf{ai\_assisted} \quad
  no signal $\Rightarrow$ \textbf{none}
};
\node[outbox, right=0.7cm of p3] (o3) {
  \texttt{corpus\_detected\_full.csv}\\[3pt]
  \textbf{60,129 repos}\\[2pt]
  agentic /\\ai\_assisted / none
};
\draw[sidarr] (p3.east) -- (o3.west);
\draw[arr=c3bd] (p2.south) -- (p3.north);

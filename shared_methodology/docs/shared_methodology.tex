\section*{\textcolor{stageblue}{1. Script Pipeline (Stages 2--5)}}

\noindent\textbf{Run order:} Stage 2 $\to$ Stage 3 $\to$ Stage 4 $\to$ Stage 5. Each stage reads
the output of the previous stage. Set \texttt{GITHUB\_TOKEN} before all API scripts.

\vspace{4pt}
\noindent\textbf{\underline{Stage 2 --- Owner Location Inference}}

\begin{tabular}{@{}lp{14cm}@{}}
\toprule
Script & What it does \\
\midrule
\cellcolor{stagegray}\texttt{2a\_location\_fast} &
  Fetches GitHub profile \texttt{location} field via GraphQL in batches of 50.
  Matches raw string against a country lookup table.
  Outputs \texttt{owner\_locations\_fast.csv} + unresolved owners for 2b. \\[2pt]
\cellcolor{stagegray}\texttt{2b\_location\_deep} &
  Deeper inference for owners with no profile location.
  Uses bio text, TLD of website, follower/following network country signals.
  Outputs enriched \texttt{owner\_locations\_deep.csv}. \\[2pt]
\cellcolor{stagegray}\texttt{2c\_timezone\_readme} &
  Applies timezone offsets + README language scripts as final signals.
  Uses Nominatim geocoding on any place names found in bio/location.
  Appends a third pass to unresolved owners. \\[2pt]
\cellcolor{stagegray}\texttt{2d\_company\_orgs} &
  Resolves company handles and GitHub org names against the ROR database.
  Provides Pass 4 --- highest precision, lowest recall.
  Merges all four passes into \texttt{owner\_locations\_all.csv}. \\
\bottomrule
\end{tabular}

  .continue/config.json, .codeium/config.json, .replit, .deepsource.toml,
  .pr\_agent.toml, .junie/, .sourcery.yaml, .kiro/, .opencode/, coderabbit.yaml,
  .coderabbit/, sweep.yaml, .sweep/, .ellipsis.yaml, .devin/, AGENT.md,
  .amazonq/, .codegen/, .goose/, .marscode/, .trae/, .speckit/, .serena/,
  .mistral/, .letta/, .abacus/, .paperclip/, .kimi/, .rovodev/, .specstory/,
  .augment/, .gru/, .pi/, .qwen/, .plandex/. \\[2pt]
\cellcolor{stagegray}\texttt{3b\_commit\_trailers} &
  Fetches latest 100 commits + 50 PRs per repo (REST).
  Checks commit \emph{messages} (trailer/body patterns), commit \emph{author names}
  (Aider \texttt{(aider)} suffix), and PR \emph{branch names} (agent prefixes).
  Tiered rule: any hit = STRONG $\to$ \texttt{agentic}. No \texttt{ai\_assisted} output.
  New patterns: \texttt{Replit-Commit-Author:}, \texttt{Generated by Replit},
  \texttt{Generated by Codex}, \texttt{codex-cli}, \texttt{(aider)} author suffix;
  new branches: \texttt{replit/}, \texttt{amazon-q/}, \texttt{pr-agent/}.
  Repos with zero signals $\to$ 3c. \\[2pt]
\cellcolor{stagegray}\texttt{3c\_deep\_scan} &
  Full tiered classification (GraphQL batch=5): 150 commits + 50 PRs + .gitignore
  + repo topics + 6 AI ignore files + 7 MEDIUM config paths + CI workflow YAMLs
  --- all in one batched query.
  New signals: \texttt{ci\_ai\_action} (12 AI GitHub Actions scanned from
  \texttt{.github/workflows/}); 10 new bot accounts (Amazon Q, Ellipsis, Greptile,
  Replit, PR-Agent, DeepSource, alternate names for Jules/Factory/Codegen/OpenHands);
  Aider author-name suffix; Replit/Codex attribution patterns.
  Produces final \texttt{corpus\_detected\_full.csv} (merged Pass A + B + C). \\
\bottomrule
\end{tabular}

\vspace{5pt}
\noindent\textbf{\underline{Stage 4 --- Geographic \& Descriptive Analysis}}

\begin{tabular}{@{}lp{14cm}@{}}
\toprule
Script & What it does \\
\midrule
\cellcolor{stagegray}\texttt{4a\_geographic} &
  Joins corpus with owner locations; computes per-country adoption rates.
  Applies Wilson 95\% CIs; excludes countries with $<$30 repos.
  Reads \texttt{corpus\_detected\_full.csv} (stage3d) + \texttt{owner\_locations\_all.csv} (stage2).
  Output: \texttt{corpus\_with\_country.csv} + \texttt{country\_adoption.csv}. \\[2pt]
\cellcolor{stagegray}\texttt{4b\_statistical} &
  Runs all three RQ significance tests (see \S 3).
  Binomial test (RQ1), McNemar (RQ2), chi-squared + pairwise Fisher (RQ3).
  Outputs \texttt{statistical\_tests.txt}. \\[2pt]
\cellcolor{stagegray}\texttt{4c\_breakdowns} &
  Adoption by ecosystem, language, and three size approaches (stars/commits/code).
  Co-adoption: tool loyalty, pairwise co-occurrence, dominant tool share.
  \textbf{NOTE:} Ecosystem columns are stack-specific --- adapt for your stack. \\[2pt]
\cellcolor{stagegray}\texttt{4d\_size\_analysis} &
  Splits corpus into tertiles (P33/P66) for three size metrics.
  Computes adoption rate per stratum; Kruskal-Wallis significance test.
  Output: \texttt{size\_comparison.csv}. \\[2pt]
\cellcolor{stagegray}\texttt{4e\_mena} &
  Replicates the full analysis for the MENA region only (20 countries).
  Sub-region breakdowns: Gulf, Levant, N.\ Africa, Turkey, Iran.
  Fisher exact vs.\ global + within-MENA chi-squared + BH correction. \\[2pt]
\cellcolor{stagegray}\texttt{4f\_pricing} &
  Classifies each detected agent as paid-dominant / free-accessible.
  Assigns World Bank 2024 income group per country.
  Tests whether tool-pricing preference varies by income group. \\
\bottomrule
\end{tabular}

\noindent\textbf{\underline{Stage 5 --- Extended Analysis}}

\begin{tabular}{@{}lp{14cm}@{}}
\toprule
Script & What it does \\
\midrule
\cellcolor{stagegray}\texttt{5a\_time\_series} &
  Monthly adoption rate from corpus creation dates (stacked area chart).
  Agent market share per month among agentic repos.
  Annotates Claude Code (Feb 2025), Codex CLI (Apr 2025), Gemini CLI (Jun 2025) releases. \\[2pt]

  Reports rank-biserial effect size $r_{rb}$ for Agentic vs.\ No AI. \\[2pt]
\cellcolor{stagegray}\texttt{5c\_fw\_subset} &
  Filters corpus to a framework subset (default: React + Next.js, no Express).
  Re-runs adoption, agent breakdown, country ranking on the subset.
  \textbf{NOTE:} Replace \texttt{FRAMEWORK\_INCLUDE/EXCLUDE} for your stack. \\[2pt]
\cellcolor{stagegray}\texttt{5d\_first\_commit} &
  For Claude Code agentic repos with trailers: finds the first trailer commit (Signal A)
  and first config file commit (Signal B) via GitHub API with checkpoint/resume.
  Classifies adoption pattern: ai\_native / config\_first / trailer\_first / trailer\_only. \\[2pt]
\cellcolor{stagegray}\texttt{5e\_country\_timeline} &
  Tracks AI coding agent adoption across top countries month by month (Feb 2025 onward).
  Produces three figures: (A) heatmap of monthly adoption rate per country, (B) bump chart
  of monthly rank by adoption rate, (C) cumulative agentic repo growth per country.
  Eligibility: $\geq$30 total repos per country; cells shown only if $\geq$5 repos that month. \\[2pt]
\cellcolor{stagegray}\texttt{5f\_tool\_loyalty} &
  Repo-level agent diversity: counts distinct agents detected per agentic repository.
  Single-agent repo = exactly 1 agent detected; multi-agent repo = 2+ agents.
  Produces: (A) stacked bar of 1 / 2 / 3+ agent buckets, (B) co-occurrence heatmap of top
  agents (only if multi-agent rate $>$5\%).
  \textbf{Limitation:} a multi-agent repo may reflect team diversity (different contributors
  using different tools) rather than one developer using multiple tools in parallel.
  Report as repo-level agent diversity, not individual tool loyalty. \\[2pt]
\cellcolor{stagegray}\texttt{5g\_world\_regions} &
  Adoption analysis across 10 macro-regions (North America, Europe, East Asia, South Asia,
  Latin America, MENA, Sub-Saharan Africa, Southeast Asia, Oceania, Central/Eastern Europe).
  Wilson 95\% CI per region, chi-squared omnibus test, pairwise Fisher + BH FDR correction.
  Reports top agent per region and single-homing rate per region. \\[2pt]
\cellcolor{stagegray}\texttt{5h\_commit\_density} &
  Measures what fraction of commits in each agentic repo were made by an AI agent
  (\texttt{ai\_density = trailer\_count / total\_commits}).
  Produces: (A) histogram of AI commit density (5\% bins) for repos with $\geq$1 trailer,
  (B) median density by agent (horizontal bar, IQR error bars),
  (C) adoption intensity buckets: config-only / low (1--25\%) / medium (26--75\%) /
  high (76--99\%) / full (100\%). \\

\end{tabular}

% ── SECTION 2: DETECTION SIGNALS ─────────────────────────────────────────────
\section*{\textcolor{stageblue}{2. Detection Signals \& Tiered Classification}}

\vspace{2pt}
\noindent\textbf{Final label rules:}
\begin{itemize}[leftmargin=1.2em,itemsep=0pt,topsep=1pt]
  \item \textbf{Agentic:} $\geq$1 STRONG family, OR $\geq$2 MEDIUM families, OR $\geq$1 MEDIUM + agent-attributed commit
  \item \textbf{AI-assisted:} only MEDIUM or only WEAK signals (no STRONG)
  \item \textbf{None:} zero signals detected across all passes
\end{itemize}

% ── SECTION 3: ANALYSIS POINTS ───────────────────────────────────────────────
\section*{\textcolor{stageblue}{3. Analysis Points (all members must report)}}

\noindent\textbf{\underline{Phase 1 --- Corpus \& Detection (Stages 1--3)}}
\begin{enumerate}[leftmargin=1.5em,itemsep=1pt,topsep=2pt]
  \item \textbf{Corpus overview:} total repo count; breakdown by package ecosystem
        (npm / pnpm / yarn / vanilla / deno / bun and equivalents for your stack);
        language split (\% TypeScript vs JavaScript, or your stack's languages).
        \textit{Visual: stacked bar --- \texttt{fig\_ecosystem\_stacked}}
  \item \textbf{Location inference:} total unique owners; \% located after each of the 4 passes
        (cumulative); top 3 countries by owner count; \% left as \texttt{not\_specified}.
        \textit{Visual: table only}
  \item \textbf{Detection pass-by-pass:} for each of the three Stage 3 passes --- input repo count,
        number classified as agentic, \% yield. Confirm each pass reduces the unclassified pool.
        \textit{Visual: grouped bar --- \texttt{fig\_pass\_contributions}}
  \item \textbf{Top agents per pass:} Pass A (config files): top 3 tools with \% share.
        Pass B (commit trailers): top tools. Pass C (deep scan): top tools.
        \textit{Visual: horizontal bar --- \texttt{fig\_tool\_distribution}}
  \item \textbf{Final classification:} agentic / ai\_assisted / none counts and \%;
        combined adoption rate (agentic + ai\_assisted) and conservative agentic-only rate.
        \textit{Visual: pipeline flow --- \texttt{fig\_pipeline\_flow}}
  \item \textbf{Adoption by language and ecosystem:} agentic \% for each primary language;
        agentic \% for each ecosystem (pnpm, npm, vanilla, yarn, deno, bun, etc.).
        \textit{Visual: stacked bar --- \texttt{fig\_ecosystem\_stacked}}
  \item \textbf{Tool loyalty and co-adoption:} \% of adopters using exactly 1 tool;
        top single tool and its share; top tool pair and repo count.
        \textit{Visual: pie + heatmap --- \texttt{fig\_tool\_loyalty}}
\end{enumerate}

\vspace{4pt}
\noindent\textbf{\underline{Phase 2 --- Geographic \& Statistical Analysis (Stage 4)}}
\begin{enumerate}[leftmargin=1.5em,itemsep=1pt,topsep=2pt]
  \item \textbf{Geographic coverage:} number of countries with $\geq$30 repos; \% of corpus
        with an assigned country; global adoption rate restricted to located repos only.
        \textit{Visual: table only}
  \item \textbf{Top 10 / Bottom 10 countries} ranked by adoption rate with Wilson 95\% CI;
        note which rates are significantly different from the global mean (BH corrected).
        \textit{Visual: dot-CI plot --- \texttt{fig\_top\_bottom\_countries}}
  \item \textbf{RQ1 --- Scope:} binomial test of global adoption vs $H_0{:}\ p \leq 5\%$;
        report rate, Wilson 95\% CI, and $p$-value.
        \textit{Visual: single number / table only}
  \item \textbf{RQ2 --- Multi-signal vs single-signal:} compare Pass-A-only rate against
        full-pipeline rate; McNemar paired test; report $\Delta$pp and $\chi^2$ statistic.
        \textit{Visual: table only}
  \item \textbf{RQ3 --- Geographic variation:} $\chi^2$ test across all eligible countries;
        pairwise Fisher exact (BH FDR corrected); report how many countries deviate
        significantly from the global mean.
        \textit{Visual: choropleth + bubble --- \texttt{fig\_choropleth}, \texttt{fig\_country\_bubble}}
  \item \textbf{Three sizing operationalisations} (all three must be reported):
        (A) stars $\leq$50 / 51--500 / $>$500;\;
        (B) commits tertiles (P33/P66);\;
        (C) code volume tertiles (P33/P66).
        For each: adoption rate per stratum, range in pp, Kruskal-Wallis significance.
        \textit{Visual: grouped bar --- \texttt{fig\_size\_grouped\_bar}}
  \item \textbf{MENA regional analysis:} define 5 sub-regions (Gulf, Levant, Turkey,
        North Africa, Iran); report repos, adoption rate and 95\% CI per sub-region and
        per eligible country ($\geq$30); Fisher exact vs global mean (BH corrected);
        within-MENA pairwise $\chi^2$.
        \textit{Visual: horizontal bar --- \texttt{fig\_mena\_bar}}
  \item \textbf{Framework analysis:} top 20 frameworks by repo count with agentic \%,
        ai\_assisted \%, and no-AI \%; note highest and lowest agentic frameworks.
        Also report 2 mutually exclusive framework combinations (e.g.\ React+Next.js
        vs React+Express) with the same breakdown.
        \textit{Visual: horizontal bar --- \texttt{fig\_framework\_adoption}}
  \item \textbf{Pricing analysis:} classify each detected agent as paid-dominant or
        free-accessible (use the classification table in \S 2). Assign World Bank 2024
        income group (HIC / UMIC / LMIC / LIC) per country. Report adoption rate and
        paid vs free tool share for each income group.
        \textit{Visual: grouped bar --- \texttt{fig\_pricing\_income}}
\end{enumerate}

\vspace{4pt}
\noindent\textbf{\underline{Phase 3 --- Extended Analysis (Stage 5)}}
\begin{enumerate}[leftmargin=1.5em,itemsep=1pt,topsep=2pt]
  \item \textbf{Time series (5a):} monthly adoption rate (stacked area chart, agentic +
        ai\_assisted) and agent market share (line chart) over the full corpus date range.
        Annotate Claude Code (Feb 2025), Codex CLI (Apr 2025), Gemini CLI (Jun 2025) releases.
        Report: peak adoption month, trough month, $\Delta$pp from first to last month,
        and the sharpest growth interval.
        \textit{Visual: stacked area + line --- \texttt{fig\_adoption\_time\_series}, \texttt{fig\_agent\_market\_share}}
  \item \textbf{Repo size comparison (5b):} median commits, code KB, stars, and forks for
        each adoption group (No AI / AI-Assisted / Agentic). Violin plots capped at 99th
        percentile. Kruskal-Wallis (3-way) + Mann-Whitney U (Agentic vs No AI) with
        rank-biserial $r_{rb}$ effect size for each metric.
        \textit{Visual: violin plots --- \texttt{fig\_size\_comparison}}
  \item \textbf{Framework subset (5c):} pick 1--2 dominant frameworks in your stack and
        re-run adoption rate, top-agent breakdown, and country ranking on that subset.
        Report overall subset adoption, top agents with \% of agentic repos, top 10
        countries by adoption rate. Compare all three against full-corpus numbers.

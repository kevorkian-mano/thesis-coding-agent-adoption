\section{Gap Analysis}
%%=============================================================================

\subsection{Signals in Paper NOT in Current}
\begin{itemize}
  \item \textbf{PR-linked commit tagging} — Paper tags every commit inside an AI-flagged PR, even with no commit-level marker. Current does not implement this. This is essential for agents like Codex that use PRs but do not sign commits.
  \item \textbf{Wider file heuristics} — Paper applies 93 file patterns; Current applies 58. The 35 missing patterns cover agents listed in Section 3.2.
  \item \textbf{Wider author heuristics} — Paper applies 79 author patterns; Current applies $\approx$28.
  \item \textbf{Per-file statistics} — Paper collects first-add date, LOC, and commit count per agent file (via shallow clone). Current does not collect this during detection.
  \item \textbf{Adoption date at scan time} — Paper records the first signal date during the scan itself. Current captured this retroactively in separate post-hoc passes.
\end{itemize}

\subsection{Signals in Current NOT in Paper}
\begin{itemize}
  \item \textbf{CI/GitHub Actions YAML scan} — Current checks \texttt{.github/workflows/} YAML files for 14 AI coding-agent GitHub Actions (e.g.\ \texttt{anthropic/claude-code-action}, \texttt{sweep-ai/sweep}).
  \item \textbf{PR body keyword matching} — Current matches 9 keyword patterns in PR description text (e.g.\ ``claude code'', ``generated by devin'', ``aider.chat''). Paper explicitly excludes PR body.
  \item \textbf{Repo topic matching} — Current uses 15 GitHub topic tags (e.g.\ \texttt{claude-code}, \texttt{agentic-ai}, \texttt{kiro}). Paper explicitly excludes repo topics.
  \item \textbf{.gitignore ``mixed'' patterns} — Current additionally flags 6 editor-mode entries (\texttt{.cursor/}, \texttt{.windsurf/}, \texttt{.codeium/}, \texttt{.continue/}, \texttt{.tabnine}, \texttt{.copilot/}).
\end{itemize}
